-- Autofarm Script for a Generic Game (Beast, Mob, Boss, etc.)
-- Note: This is a template and needs to be adapted to the specific game's API and environment.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer
local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

-- Configuration
local config = {
    autoFarm = true,
    autoSearchBeast = true,
    autoMob = true,
    autoBoss = true,
    searchRadius = 1000,
    attackDistance = 10,
    attackCooldown = 1,
    teleportToTarget = true,
    debug = false,
}

-- Utility Functions
local function debugPrint(...)
    if config.debug then
        print(...)
    end
end

local function getNearestTarget(targetType)
    -- This function should be customized based on the game's NPC structure
    local nearestTarget = nil
    local nearestDistance = math.huge

    for _, npc in pairs(workspace:GetChildren()) do
        if npc:IsA("Model") and npc:FindFirstChild("Humanoid") and npc:FindFirstChild("HumanoidRootPart") then
            local humanoid = npc.Humanoid
            local hrp = npc.HumanoidRootPart

            if humanoid.Health > 0 then
                local distance = (HumanoidRootPart.Position - hrp.Position).Magnitude
                if distance <= config.searchRadius then
                    -- Example filters for target types (customize as needed)
                    local name = npc.Name:lower()
                    local isTarget = false
                    if targetType == "beast" and name:find("beast") then
                        isTarget = true
                    elseif targetType == "mob" and name:find("mob") then
                        isTarget = true
                    elseif targetType == "boss" and name:find("boss") then
                        isTarget = true
                    elseif targetType == "any" then
                        isTarget = true
                    end

                    if isTarget and distance < nearestDistance then
                        nearestDistance = distance
                        nearestTarget = npc
                    end
                end
            end
        end
    end

    return nearestTarget
end

local function attackTarget(target)
    if not target or not target:FindFirstChild("HumanoidRootPart") then return end
    local hrp = target.HumanoidRootPart

    if config.teleportToTarget then
        HumanoidRootPart.CFrame = hrp.CFrame * CFrame.new(0, 0, -config.attackDistance)
    end

    -- Simulate attack (customize based on game mechanics)
    -- Example: Fire a remote event or use tool activation
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool") then
        local tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
        tool:Activate()
    end

    debugPrint("Attacking target:", target.Name)
end

-- Main Autofarm Loop
spawn(function()
    while true do
        if config.autoFarm then
            local target = nil
            if config.autoBoss then
                target = getNearestTarget("boss")
            end
            if not target and config.autoBeast then
                target = getNearestTarget("beast")
            end
            if not target and config.autoMob then
                target = getNearestTarget("mob")
            end
            if not target then
                target = getNearestTarget("any")
            end

            if target then
                attackTarget(target)
            end
        end
        wait(config.attackCooldown)
    end
end)

-- User Interface (Basic Toggle with Keyboard)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F then
        config.autoFarm = not config.autoFarm
        print("AutoFarm toggled:", config.autoFarm)
    end
end)

print("Autofarm script loaded. Press 'F' to toggle autofarm.")
